function saveAssessment(e) { e.preventDefault(); let adlScores = []; let total = 0; for(let i=1; i<=10; i++) { const radios = document.getElementsByName(`adl_${i}`); let val = 0; for(let r of radios) { if(r.checked) { val = parseInt(r.value); break; } } adlScores.push(val); total += val; } const recordData = { id: document.getElementById('edit-id').value || 'LTC-' + Date.now(), staffName: document.getElementById('staff-name').value, staffArea: document.getElementById('staff-area').value, assessmentDate: document.getElementById('assessment-date').value, patientName: document.getElementById('patient-name').value, patientCid: document.getElementById('patient-cid').value, patientPhone: document.getElementById('patient-phone').value, adlScores: JSON.stringify(adlScores), totalScore: total, ltcGroup: getLTCGroup(total).name, updatedAt: new Date().toISOString() }; const submitBtn = document.getElementById('submit-btn'); submitBtn.disabled = true; submitBtn.innerHTML = ' กำลังส่งข้อมูลเข้า Google Sheets...'; // เทคนิคสร้าง Form ส่งข้อมูลข้ามเซิร์ฟเวอร์แบบไม่ติด CORS let iframe = document.getElementById('hidden_submit_iframe'); if(!iframe) { iframe = document.createElement('iframe'); iframe.id = 'hidden_submit_iframe'; iframe.name = 'hidden_submit_iframe'; iframe.style.display = 'none'; document.body.appendChild(iframe); } const form = document.createElement('form'); form.method = 'POST'; form.action = gasUrl; form.target = 'hidden_submit_iframe'; for (let key in recordData) { const input = document.createElement('input'); input.type = 'hidden'; input.name = key; input.value = recordData[key]; form.appendChild(input); } document.body.appendChild(form); form.submit(); document.body.removeChild(form); setTimeout(() => { showToast("บันทึกข้อมูลเข้า Google Sheets กลางเรียบร้อย!", "success"); submitBtn.disabled = false; submitBtn.innerHTML = ' บันทึกข้อมูลขึ้น Google Sheets กลาง'; resetForm(); switchTab('records'); fetchCentralRecords(false); }, 2000); }
